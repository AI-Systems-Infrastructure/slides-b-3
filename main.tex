  \begin{block}{Software deployment without containers}
    \begin{itemize}
      \item Setting up everything right on each computer the software is deployed to is tedious and prone to error
      \item Multiple pieces of software running on the same computer might interfere with each other
    \end{itemize}

    \begin{itemize}
      \item A container \textbf{packs everything a piece of software needs in one self-contained package}
      \item A container provides an isolated environment for software to run without interference
    \end{itemize}
  \end{block}

\section{Building Images}

\begin{frame}[fragile]{The interactive approach}
  Docker allows for committing a container instance into an image.

    \begin{minted}{bash}
docker run -it -v $(pwd):/app --name ais-builder python:3.13 bash

# Inside the container instance:
pip install fastapi uvicorn
mkdir -p /opt/ai-system && cp /app/main.py /opt/ai-system/
exit

docker commit ais-builder ai-system:1.0
    \end{minted}

\note{
}
\end{frame}


\begin{frame}{The interactive approach}{Looks easy but a bad approach}
  \texttt{docker commit} will squash everything we did to the instance into one opaque image layer.

  \begin{itemize}
    \item \textbf{No record of the setup of our image:} prone to memory loss, challenges future revision to the image
    \item \textbf{Throws away the layered structure:} breaks one of the major benefits of container images
  \end{itemize}

\end{frame}

\begin{frame}{Building images using Dockerfile}
  Document the building process of an image in a \textbf{Dockerfile}.
  Docker can consistently build the image given its Dockerfile and the build context.

\begin{frame}{Building images using Dockerfile}{Usually a good practice}
  \begin{itemize}
    \item \textbf{Clear record of the setup of our image:} Dockerfile serves as the record of the setup process, no worry about forgetting and easy to revise
    \item \textbf{Corresponds to the layered structure:} each command in the Dockerfile corresponds to one image layer in the built image
  \end{itemize}

  % TODO: Embed a diagram showing the correspondence between Dockerfile and image layers
\note{
}
\end{frame}


\begin{frame}[fragile]{Building images using Dockerfile}{Build command}
  \begin{minted}{bash}
docker build -t <name>[:<tag>] [-f <dockerfile>] <context>
  \end{minted}
  \begin{itemize}
    \item \texttt{-t <name>[:<tag>]}: name and tag of the image, tag defaults to \texttt{latest}
    \item \texttt{-f <dockerfile>}: Dockerfile to use, defaults to \texttt{Dockerfile} in the build context
    \item \texttt{<context>}: the \textbf{build context}, directory sent to Docker, the only files the build can access
  \end{itemize}

  \vspace{1em}

  \begin{minted}{bash}
docker build -t ai-system:1.0 .
  \end{minted}
  \begin{itemize}
    \item Build \texttt{ai-system:1.0} with the current directory as the build context
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}{What is a Dockerfile?}
  Plain-text definition of how a container image is built, each line starts with a specific command.

  \begin{itemize}
      \item \texttt{FROM <image>}: pick the base image, always first
      \item \texttt{WORKDIR <path>}: set the working directory inside the image
      \item \texttt{COPY <src> <dest>}: bring files in from the build context
      \item \texttt{RUN <cmd>}: execute a shell command during the build
      \item \texttt{ENV KEY=value}: set an environment variable visible at runtime
      \item \texttt{EXPOSE <port>}: document the port, does not actually publish it
      \item \texttt{CMD ["program", "arg"]}: default command when a container starts
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{FROM}
  \begin{minted}{dockerfile}
FROM <image>[:<tag>]
  \end{minted}
  \begin{itemize}
    \item \texttt{<image>}: name of the base image to build on top of, e.g., one from Docker Hub
    \item \texttt{<tag>}: version of the base image, defaults to \texttt{latest} if omitted
    \item Comes first in a Dockerfile, every later command builds on top of it
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
FROM python:3.13-slim
  \end{minted}
  \begin{itemize}
    \item Start from the slim variant of the official Python 3.13 image
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{WORKDIR}
  \begin{minted}{dockerfile}
WORKDIR <path>
  \end{minted}
  \begin{itemize}
    \item \texttt{<path>}: directory inside the image, created if it does not exist
    \item Applies to all later commands, relative paths resolve against it
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
WORKDIR /app
  \end{minted}
  \begin{itemize}
    \item Later commands run in \texttt{/app}, and \texttt{.} as a destination refers to \texttt{/app}
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{COPY}
  \begin{minted}{dockerfile}
COPY <src> [<src> ...] <dest>
  \end{minted}
  \begin{itemize}
    \item \texttt{<src>}: one or more files or directories from the build context
    \item \texttt{<dest>}: path inside the image, relative to the working directory unless absolute
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
COPY requirements.txt .
  \end{minted}
  \begin{itemize}
    \item Copy \texttt{requirements.txt} from the build context into the working directory
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{RUN}
  \begin{minted}{dockerfile}
RUN <command>
  \end{minted}
  \begin{itemize}
    \item \texttt{<command>}: shell command executed at build time, not when the container starts
    \item Changes it makes to the file system are saved as a new image layer
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
RUN pip install --no-cache-dir -r requirements.txt
  \end{minted}
  \begin{itemize}
    \item Install the Python packages into the image, without keeping pip's download cache
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{ENV}
  \begin{minted}{dockerfile}
ENV <key>=<value> [<key>=<value> ...]
  \end{minted}
  \begin{itemize}
    \item \texttt{<key>=<value>}: name and value of an environment variable, multiple pairs allowed
    \item Visible to later commands during the build and to the running container
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
ENV DATABASE_URL=sqlite:////app/data/ai_api.db
  \end{minted}
  \begin{itemize}
    \item Default database location for the server, can be overridden with \texttt{docker run -e}
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{EXPOSE}
  \begin{minted}{dockerfile}
EXPOSE <port>[/<protocol>]
  \end{minted}
  \begin{itemize}
    \item \texttt{<port>}: port the containerized program listens on
    \item \texttt{<protocol>}: \texttt{tcp} or \texttt{udp}, defaults to \texttt{tcp} if omitted
    \item Documentation only, publishing the port still needs \texttt{docker run -p}
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
EXPOSE 8000
  \end{minted}
  \begin{itemize}
    \item Declare that the API server listens on port 8000 inside the container
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Dockerfile commands}{CMD}
  \begin{minted}{dockerfile}
CMD ["<program>", "<arg1>", "<arg2>"]
  \end{minted}
  \begin{itemize}
    \item \texttt{<program>}, \texttt{<argN>}: program to run and any number of arguments, as a JSON array
    \item Default command when a container starts, replaced by any command given to \texttt{docker run}
    \item Only the last \texttt{CMD} in a Dockerfile takes effect
  \end{itemize}

  \vspace{1em}

  \begin{minted}{dockerfile}
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
  \end{minted}
  \begin{itemize}
    \item Start the FastAPI server with Uvicorn, listening on port 8000
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Command order in a Dockerfile}
  As soon as one layer's input changes, every layer on top of it has to be rebuilt as well.

    \begin{block}{Worse order}
        \begin{minted}{dockerfile}
COPY . .
RUN pip install -r requirements.txt
        \end{minted}
        Every edit to \texttt{main.py} changes the input of \texttt{COPY}, which forces its subsequent layers (\texttt{pip install}) to rerun.
    \end{block}

    \begin{block}{Better order}
        \begin{minted}{dockerfile}
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .
        \end{minted}
        \texttt{pip install} only reruns when \texttt{requirements.txt} changes.
    \end{block}
\note{
}
\end{frame}



\section{Containerizing AI System}

\begin{frame}{Project layout}
  Suppose we migrate our API server from Module A.4 into the following project structure.
  \begin{itemize}
      \item \texttt{main.py} and other Python scripts of the server
      \item \texttt{requirements.txt} listing the Python packages we need
      \item the \texttt{Dockerfile} we are about to write
  \end{itemize}
\note{
}
\end{frame}


\begin{frame}[fragile]{Our Dockerfile}
    \begin{minted}{dockerfile}
FROM python:3.13-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY main.py ./

EXPOSE 8000

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
    \end{minted}

    \begin{itemize}
      \item By default, our server only accepts requests from \texttt{localhost}
      \item The \texttt{--host 0.0.0.0} argument makes our server accept requests from clients of all IP addresses
      \item Even with port mapping, requests from outside the container instance won't be from \texttt{localhost}
    \end{itemize}
    
\note{
}
\end{frame}


\begin{frame}[fragile]{Build and run}
Build the container image.
    \begin{minted}{bash}
docker build -t my-ai-server:1.0 .
    \end{minted}

Then run it with port mapping.
    \begin{minted}{bash}
docker run -d --name ai-server -p 8000:8000 my-ai-server:1.0
    \end{minted}

Now \texttt{http://127.0.0.1:8000/v1/classify} should be reachable from the host computer.
\note{
}
\end{frame}


\begin{frame}[fragile]{Distribute our image to Docker Hub}
  Like GitHub repositories, container images can be pushed to Docker Hub (or other container registries).

    \begin{minted}{bash}
# 1. Log in
docker login

# 2. Tag the local image with the registry path
# Tag format: [<registry>/]<namespace>/<repository>:<tag>
docker tag my-ai-server:1.0 yourusername/my-ai-server:1.0

# 3. Push
docker push yourusername/my-ai-server:1.0
    \end{minted}

    Anyone can now pull our image using \texttt{docker pull yourusername/my-ai-server:1.0}.
\note{
}
\end{frame}



\section{Layers vs Volumes}

\begin{frame}{The recurring design choice}
    For each file, we have to decide: inside the image as a layer, or mounted in from the outside as a volume?

    \begin{block}{General principle}
\begin{itemize}
\item \textbf{Inside:} things that should be the same across every running instance
\item \textbf{Volume:} things that should differ between instances and/or persist beyond one instance
\end{itemize}
    \end{block}

    \begin{block}{Examples}
    \begin{itemize}
        \item Application code and Python packages: inside, since every instance runs the same
        \item User database: volume, differ between instances and should survive container restart
        \item Model weights: can be inside or volume or even non-persistent, each with caveats
    \end{itemize}
    \end{block}
\note{
}
\end{frame}


\begin{frame}[fragile]{Baking model weights into the image}
    For small models, download at build time is fine:
    \begin{minted}{dockerfile}
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Pre-download ResNet-18 weights into a layer
RUN python -c "from torchvision.models import \
                   resnet18, ResNet18_Weights; \
               resnet18(weights=ResNet18_Weights.DEFAULT)"

COPY main.py db.py ./
    \end{minted}

    A fresh container now starts immediately, with no runtime download.
    For very large models, consider falling back to downloading at startup.
\note{
}
\end{frame}


\begin{frame}[fragile]{Persisting the database via a volume}
    The SQLite database file stores our users, API keys, and request logs, and it must survive container removal.
    If the database lives inside the writable layer, removing the container deletes all our data.

    \begin{minted}{bash}
mkdir -p data

docker run -d --name ai-server \
    -p 8000:8000 \
    -v $(pwd)/data:/app/data \
    -e DATABASE_URL=sqlite:////app/data/ai_api.db \
    my-ai-server:1.0
    \end{minted}

    \begin{itemize}
        \item Database lives on the host, survives container removal
        \item Server reads \texttt{DATABASE\_URL} from the environment
        \item We can swap the container for a newer image and keep the data
    \end{itemize}
\note{
}
\end{frame}



\section{GPU Passthrough}

\begin{frame}{Containers do not see the GPU by default}
    If we run our server on a machine with an NVIDIA GPU, the container needs explicit access to the GPU device.
    Containers are isolated from host hardware, and the GPU has to be passed through.

    \begin{itemize}
        \item \texttt{--device <host>} grants access to one specific device file
        \item Same pattern works for webcams, serial ports, USB, sound cards
        \item \texttt{--privileged} drops all device restrictions at once, but strips away most of the isolation containers exist to provide
    \end{itemize}

    GPUs need more than raw device files. They also need userspace driver libraries that match the kernel module on the host.
\note{
}
\end{frame}


\begin{frame}[fragile]{NVIDIA Container Toolkit}
    The standard tool for NVIDIA GPUs.
    Handles the userspace libraries and bridges the host driver to the container.

    \begin{minted}{bash}
docker run -d --name ai-server \
    --gpus all \
    -p 8000:8000 \
    my-ai-server:1.0
    \end{minted}

    \begin{itemize}
        \item \texttt{--gpus all}: every GPU on the host
        \item \texttt{--gpus "device=0"}: only one specific GPU
        \item \texttt{--gpus 2}: limit to two GPUs
    \end{itemize}
\note{
}
\end{frame}


\begin{frame}{Other AI hardware}
    The \texttt{--gpus} flag only covers NVIDIA.

    \begin{itemize}
        \item AMD GPUs use ROCm. Access via \texttt{--device /dev/kfd /dev/dri}, or AMD's container toolkit
        \item Google TPUs require some more complicated setup
        \item Consumer NPUs (Apple Neural Engine, Qualcomm, AMD Ryzen AI) mostly are not exposed to containers
    \end{itemize}
\note{
}
\end{frame}
